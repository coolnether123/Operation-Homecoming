    public void Notify_Explosion(Explosion explosion)

        return;

    }

    private IEnumerable<Pawn> GetPackmates(Pawn pawn, float radius)
    {
      District pawnRoom = pawn.GetDistrict();
      IReadOnlyList<Pawn> raceMates = pawn.Map.mapPawns.AllPawnsSpawned;
      for (int i = 0; i < raceMates.Count; ++i)
      {
        if (pawn != raceMates[i] && raceMates[i].def == pawn.def && raceMates[i].Faction == pawn.Faction && raceMates[i].Position.InHorDistOf(pawn.Position, radius) && raceMates[i].GetDistrict() == pawnRoom)
          yield return raceMates[i];
      }
    }

    private void StartManhunterBecauseOfPawnAction(
      Pawn instigator,
      string letterTextKey,
      bool causedByDamage = false)
    {
      if (!this.mentalStateHandler.TryStartMentalState(MentalStateDefOf.Manhunter))
        return;
      string text = (string) letterTextKey.Translate((NamedArgument) this.pawn.Label, this.pawn.Named("PAWN")).AdjustedFor(this.pawn);
      GlobalTargetInfo globalTargetInfo = (GlobalTargetInfo) (Thing) this.pawn;
      float num1 = 0.5f;
      if (causedByDamage)
        num1 *= PawnUtility.GetManhunterChanceFactorForInstigator(instigator);
      int num2 = 1;
      if (Find.Storyteller.difficulty.allowBigThreats && (double) Rand.Value < (double) num1)
      {
        foreach (Pawn packmate in this.GetPackmates(this.pawn, 24f))
        {
          if (packmate.mindState.mentalStateHandler.TryStartMentalState(MentalStateDefOf.Manhunter, causedByDamage: causedByDamage))
            ++num2;
        }
        if (num2 > 1)
        {
          globalTargetInfo = (GlobalTargetInfo) new TargetInfo(this.pawn.Position, this.pawn.Map);
          text = (string) (text + "\n\n" + "AnimalManhunterOthers".Translate((NamedArgument) this.pawn.kindDef.GetLabelPlural(), (NamedArgument) (Thing) this.pawn));
        }
      }
      Find.LetterStack.ReceiveLetter((TaggedString) (string) "LetterLabelAnimalManhunterRevenge".Translate((NamedArgument) (this.pawn.IsNonMutantAnimal ? this.pawn.Label : this.pawn.def.label)).CapitalizeFirst(), (TaggedString) text, num2 == 1 ? LetterDefOf.ThreatSmall : LetterDefOf.ThreatBig, (LookTargets) globalTargetInfo);
    }

    private static bool CanStartFleeingBecauseOfPawnAction(Pawn p)
    {
      if (!p.IsNonMutantAnimal || p.InMentalState || p.IsFighting() || p.Downed || p.Dead || ThinkNode_ConditionalShouldFollowMaster.ShouldFollowMaster(p))
        return false;
      return p.jobs.curJob == null || p.jobs.curJob.def != JobDefOf.Flee || p.jobs.curJob.startTick != Find.TickManager.TicksGame;
    }

    public void StartFleeingBecauseOfPawnAction(Thing instigator)
    {
      List<Thing> threats = new List<Thing>() { instigator };
      IntVec3 fleeDest1 = CellFinderLoose.GetFleeDest(this.pawn, threats, this.pawn.Position.DistanceTo(instigator.Position) + 28f);
      if (fleeDest1 != this.pawn.Position)
      {
        IntVec3 intVec3 = fleeDest1 - this.pawn.Position;
        Vector3 vector3_1 = intVec3.ToVector3();
        intVec3 = this.pawn.Map.Center - this.pawn.Position;
        Vector3 vector3_2 = intVec3.ToVector3();
        bool flag = (double) Vector3.Dot(vector3_1, vector3_2) < 0.0;
        IntVec3 position = new IntVec3();
        if (((!this.pawn.IsNonMutantAnimal ? 0 : (this.pawn.Faction == null ? 1 : 0)) & (flag ? 1 : 0)) != 0 && Rand.Chance(0.5f) && CellFinderLoose.GetFleeExitPosition(this.pawn, 10f, out position))
        {
          Job newJob = JobMaker.MakeJob(JobDefOf.Flee, (LocalTargetInfo) position, (LocalTargetInfo) instigator);
          newJob.exitMapOnArrival = true;
          this.pawn.jobs.StartJob(newJob, JobCondition.InterruptOptional);
        }
        else
          this.pawn.jobs.StartJob(JobMaker.MakeJob(JobDefOf.Flee, (LocalTargetInfo) fleeDest1, (LocalTargetInfo) instigator), JobCondition.InterruptOptional);
      }
      if (!this.pawn.RaceProps.herdAnimal || !Rand.Chance(0.1f))
        return;
      foreach (Pawn packmate in this.GetPackmates(this.pawn, 24f))
      {
        if (Pawn_MindState.CanStartFleeingBecauseOfPawnAction(packmate))
        {
          IntVec3 fleeDest2 = CellFinderLoose.GetFleeDest(packmate, threats, packmate.Position.DistanceTo(instigator.Position) + 28f);
          if (fleeDest2 != packmate.Position)
            packmate.jobs.StartJob(JobMaker.MakeJob(JobDefOf.Flee, (LocalTargetInfo) fleeDest2, (LocalTargetInfo) instigator), JobCondition.InterruptOptional);
        }
      }
    }

    public void SetAutofeeder(Pawn feeder, AutofeedMode setting)
    {
      if (setting == AutofeedMode.Childcare)
        this.autoFeeders.Remove(feeder);
      else
        this.autoFeeders[feeder] = setting;
      if (setting != AutofeedMode.Childcare)
        return;
      if (feeder.WorkTypeIsDisabled(WorkTypeDefOf.Childcare))
      {
        Messages.Message((string) "MessageChildcareDisabled".Translate(feeder.Named("FEEDER")), (LookTargets) (Thing) feeder, MessageTypeDefOf.CautionInput, false);
      }
      else
      {
        if (feeder.workSettings.WorkIsActive(WorkTypeDefOf.Childcare))
          return;
        Messages.Message((string) "MessageChildcareNotAssigned".Translate(feeder.Named("FEEDER")), (LookTargets) (Thing) feeder, MessageTypeDefOf.CautionInput, false);
      }
    }

    public AutofeedMode AutofeedSetting(Pawn feeder)
    {
      return !this.pawn.DevelopmentalStage.Baby() ? AutofeedMode.Never : this.autoFeeders.TryGetValue<Pawn, AutofeedMode>(feeder, AutofeedMode.Childcare);
    }

    public IEnumerable<Pawn> Autofeeders()
    {
      return !ChildcareUtility.CanSuckle(this.pawn, out ChildcareUtility.BreastfeedFailReason? _) ? Enumerable.Empty<Pawn>() : (IEnumerable<Pawn>) this.autoFeeders.Keys;
    }

    public bool AnyAutofeeder(
      AutofeedMode autofeed,
      Predicate<Pawn, Pawn> feederPredicate,
      List<Pawn> possibleFeeders = null)
    {
      if (!ChildcareUtility.CanSuckle(this.pawn, out ChildcareUtility.BreastfeedFailReason? _))
        return false;
      possibleFeeders = possibleFeeders ?? this.pawn.MapHeld.mapPawns.FreeHumanlikesOfFaction(this.pawn.Faction);
      foreach (Pawn possibleFeeder in possibleFeeders)
      {
        if (possibleFeeder != this.pawn && autofeed == this.AutofeedSetting(possibleFeeder) && feederPredicate(this.pawn, possibleFeeder))
          return true;
      }
      return false;
    }

    public void ClearBreastfeedCaravan() => this.babyCaravanBreastfeed.Clear();

    public void BreastfeedCaravan(Pawn baby, float amountPct)
    {
      if (this.pawn.GetCaravan() == null)
        Log.Warning(this.pawn.Label + " tried to caravan breastfeed while not in a caravan");
      if (!this.babyCaravanBreastfeed.ContainsKey(baby))
        this.babyCaravanBreastfeed.Add(baby, 0.0f);
      float num = this.babyCaravanBreastfeed[baby] + amountPct;
      if ((double) num >= 0.60000002384185791)
      {
        num -= 0.6f;
        baby.needs.mood.thoughts.memories.TryGainMemory(ThoughtDefOf.BreastfedMe, this.pawn);
        this.pawn.needs.mood.thoughts.memories.TryGainMemory(ThoughtDefOf.BreastfedBaby, baby);
      }
      this.babyCaravanBreastfeed[baby] = num;
    }
