      if (mode != DestroyMode.Vanish && mode != DestroyMode.KillFinalize)
        Log.Error("Destroyed pawn " + (object) this + " with unsupported mode " + (object) mode + ".");
      Map mapHeld = this.MapHeld;
      base.Destroy(mode);
      Find.WorldPawns.Notify_PawnDestroyed(this);
      if (this.ownership != null)
      {
        Building_Grave assignedGrave = this.ownership.AssignedGrave;
        this.ownership.UnclaimAll();
        if (mode == DestroyMode.KillFinalize && assignedGrave != null)
          assignedGrave.CompAssignableToPawn.TryAssignPawn(this);
      }
      this.ClearMind(clearInspiration: true);
      Lord lord = this.GetLord();
      if (lord != null)
      {
        PawnLostCondition cond = mode == DestroyMode.KillFinalize ? PawnLostCondition.Killed : PawnLostCondition.Vanished;
        lord.Notify_PawnLost(this, cond);
      }
      if (Current.ProgramState == ProgramState.Playing)
      {
        Find.GameEnder.CheckOrUpdateGameOver();
        Find.TaleManager.Notify_PawnDestroyed(this);
      }
      foreach (Pawn pawn in PawnsFinder.AllMapsWorldAndTemporary_Alive.Where<Pawn>((Func<Pawn, bool>) (p => p.playerSettings != null && p.playerSettings.Master == this)))
        pawn.playerSettings.Master = (Pawn) null;
      this.equipment?.Notify_PawnDied();
      if (ModsConfig.AnomalyActive && Find.Anomaly != null)
        Find.Anomaly.Notify_PawnDied(this);
      if (mode != DestroyMode.KillFinalize)
      {
        this.equipment?.DestroyAllEquipment();
        this.inventory?.DestroyAll();
        this.apparel?.DestroyAll();
      }
      WorldPawns worldPawns = Find.WorldPawns;
      if (!worldPawns.IsBeingDiscarded(this) && !worldPawns.Contains(this))
        worldPawns.PassToWorld(this);
      if (this.Faction.IsPlayerSafe())
        this.Ideo?.RecacheColonistBelieverCount();
      this.relations?.Notify_PawnDestroyed(mode);
      this.Drawer?.renderer?.renderTree?.SetDirty();
    }

    public override void DeSpawn(DestroyMode mode = DestroyMode.Vanish)
    {
      Map map = this.Map;
      if (this.jobs?.curJob != null)
        this.jobs.StopAll();
      base.DeSpawn(mode);
      this.pather?.StopDead();
      this.roping?.Notify_DeSpawned();
      this.mindState.droppedWeapon = (Thing) null;
      this.needs?.mood?.thoughts.situational.Notify_SituationalThoughtsDirty();
      this.meleeVerbs?.Notify_PawnDespawned();
      this.mechanitor?.Notify_DeSpawned(mode);
      this.ClearAllReservations(false);
      if (map != null)
      {
        map.mapPawns.DeRegisterPawn(this);
        map.autoSlaughterManager.Notify_PawnDespawned();
      }
      PawnComponentsUtility.RemoveComponentsOnDespawned(this);
      if (this.sustainerAmbient != null)
      {
        this.sustainerAmbient.End();
        this.sustainerAmbient = (Sustainer) null;
      }
      if (this.sustainerMoving == null)
        return;
      this.sustainerMoving.End();
      this.sustainerMoving = (Sustainer) null;
    }

    public override void Discard(bool silentlyRemoveReferences = false)
    {
      if (Find.WorldPawns.Contains(this))
      {
        Log.Warning("Tried to discard a world pawn " + (object) this + ".");
      }
      else
      {
        base.Discard(silentlyRemoveReferences);
        if (this.relations != null)
        {
          if (this.RaceProps.Humanlike && this.relations.Children.Count<Pawn>((Func<Pawn, bool>) (x => !x.markedForDiscard)) > 1)
          {
            foreach (Pawn child in this.relations.Children)
            {
              if (!child.markedForDiscard)
              {
                DirectPawnRelation directRelation = child.relations.GetDirectRelation(PawnRelationDefOf.Parent, this);
                child.relations.ElevateToVirtualRelation(directRelation);
              }
            }
          }
          this.relations.ClearAllRelations();
        }
        if (Current.ProgramState == ProgramState.Playing)
        {
          Find.PlayLog.Notify_PawnDiscarded(this, silentlyRemoveReferences);
          Find.BattleLog.Notify_PawnDiscarded(this, silentlyRemoveReferences);
          Find.TaleManager.Notify_PawnDiscarded(this, silentlyRemoveReferences);
          Find.QuestManager.Notify_PawnDiscarded(this);
        }
        foreach (Pawn pawn in PawnsFinder.AllMapsWorldAndTemporary_Alive)
          pawn.needs?.mood?.thoughts.memories.Notify_PawnDiscarded(this);
        Corpse.PostCorpseDestroy(this, true);
      }
    }

    public Corpse MakeCorpse(Building_Grave assignedGrave, bool inBed, float bedRotation)
    {
      if (this.holdingOwner != null)
      {
        Log.Warning("We can't make corpse because the pawn is in a ThingOwner. Remove him from the container first. This should have been already handled before calling this method. holder=" + (object) this.ParentHolder);
        return (Corpse) null;

      if (this.RaceProps.corpseDef == null)
        return (Corpse) null;
      Corpse corpse = (Corpse) ThingMaker.MakeThing(this.RaceProps.corpseDef);

        corpse.InnerPawn.ownership.ClaimGrave(assignedGrave);

        corpse.InnerPawn.Drawer.renderer.wiggler.SetToCustomRotation(bedRotation + 180f);

    public void ExitMap(bool allowedToJoinOrCreateCaravan, Rot4 exitDir)
    {
      if (this.IsWorldPawn())
      {
        Log.Warning("Called ExitMap() on world pawn " + (object) this);
      }
      else
      {
        this.Ideo?.Notify_MemberLost(this, this.Map);
        if (allowedToJoinOrCreateCaravan && CaravanExitMapUtility.CanExitMapAndJoinOrCreateCaravanNow(this))
        {
          CaravanExitMapUtility.ExitMapAndJoinOrCreateCaravan(this, exitDir);
        }
        else
        {
          this.GetLord()?.Notify_PawnLost(this, PawnLostCondition.ExitedMap);
          if (this.carryTracker?.CarriedThing != null)
          {
            if (this.carryTracker.CarriedThing is Pawn carriedThing)
            {
              if (this.Faction != null && this.Faction != carriedThing.Faction)
              {
                this.Faction.kidnapped.Kidnap(carriedThing, this);
              }
              else
              {
                if (!this.teleporting)
                  this.carryTracker.innerContainer.Remove((Thing) carriedThing);
                carriedThing.teleporting = this.teleporting;
                carriedThing.ExitMap(false, exitDir);
                carriedThing.teleporting = false;
              }
            }
            else
              this.carryTracker.CarriedThing.Destroy();
            if (!this.teleporting || carriedThing == null)
              this.carryTracker.innerContainer.Clear();
          }
          bool flag1 = ThingOwnerUtility.AnyParentIs<ActiveDropPodInfo>((Thing) this) || ThingOwnerUtility.AnyParentIs<TravelingTransportPods>((Thing) this);
          bool flag2 = ((this.IsCaravanMember() ? 1 : (this.teleporting ? 1 : 0)) | (flag1 ? 1 : 0)) != 0;
          bool flag3 = !flag2 || !this.IsPrisoner && !this.IsSlave && !flag1 || this.guest != null && this.guest.Released;
          bool freed = flag3 && (this.IsPrisoner || this.IsSlave) && this.guest != null && this.guest.Released;
          bool flag4 = freed || this.guest != null && this.guest.HostFaction == Faction.OfPlayer;
          if (flag3 && !flag2)
          {
            foreach (Thing orInventoryThing in this.EquippedWornOrInventoryThings)
              orInventoryThing.GetStyleSourcePrecept()?.Notify_ThingLost(orInventoryThing);
          }
          this.Faction?.Notify_MemberExitedMap(this, freed);
          if (this.Faction == Faction.OfPlayer && this.IsSlave && this.SlaveFaction != null && this.SlaveFaction != Faction.OfPlayer && this.guest.Released)
            this.SlaveFaction.Notify_MemberExitedMap(this, freed);
          if (this.ownership != null & flag4)
            this.ownership.UnclaimAll();
          if (this.guest != null)
          {
            int num = this.IsPrisonerOfColony ? 1 : 0;
            if (freed)
              this.guest.SetGuestStatus((Faction) null);
            if (num != 0)
            {
              this.guest.SetNoInteraction();
              if (!this.guest.Released & flag3)
                GuestUtility.Notify_PrisonerEscaped(this);
            }
            this.guest.Released = false;
          }
          this.DeSpawnOrDeselect();
          this.inventory.UnloadEverything = false;
          if (flag3)
            this.ClearMind();
          this.relations?.Notify_ExitedMap();
          Find.WorldPawns.PassToWorld(this);
          QuestUtility.SendQuestTargetSignals(this.questTags, "LeftMap", this.Named("SUBJECT"));
          Find.FactionManager.Notify_PawnLeftMap(this);
          Find.IdeoManager.Notify_PawnLeftMap(this);
        }
      }
    }

    public override void PreTraded(TradeAction action, Pawn playerNegotiator, ITrader trader)
    {
      base.PreTraded(action, playerNegotiator, trader);
      if (this.SpawnedOrAnyParentSpawned)
        this.DropAndForbidEverything();
      this.ownership?.UnclaimAll();
      if (action == TradeAction.PlayerSells)
      {
        Faction other = this.GetExtraHomeFaction() ?? this.GetExtraHostFaction();
        if (other != null && other != Faction.OfPlayer)
          Faction.OfPlayer.TryAffectGoodwillWith(other, Faction.OfPlayer.GoodwillToMakeHostile(other), reason: HistoryEventDefOf.MemberSold, lookTarget: new GlobalTargetInfo?((GlobalTargetInfo) (Thing) this));
      }
      this.guest?.SetGuestStatus((Faction) null);
      switch (action)
      {
        case TradeAction.PlayerBuys:
          if (this.guest != null && this.guest.joinStatus == JoinStatus.JoinAsSlave)
          {
            this.guest.SetGuestStatus(Faction.OfPlayer, RimWorld.GuestStatus.Slave);
            break;
          }
          this.needs.mood?.thoughts.memories.TryGainMemory(ThoughtDefOf.FreedFromSlavery);
          this.SetFaction(Faction.OfPlayer, (Pawn) null);
          break;
        case TradeAction.PlayerSells:
          if (this.RaceProps.Humanlike)
            TaleRecorder.RecordTale(TaleDefOf.SoldPrisoner, (object) playerNegotiator, (object) this, (object) trader);
          if (this.Faction != null)
            this.SetFaction((Faction) null, (Pawn) null);
          if (this.RaceProps.IsFlesh)
          {
            this.relations.Notify_PawnSold(playerNegotiator);
            break;
          }
          break;
      }
      this.ClearMind();
    }

    public override void SetFaction(Faction newFaction, Pawn recruiter = null)
    {
      if (newFaction == this.Faction)
      {
        Log.Warning("Used SetFaction to change " + this.ToStringSafe<Pawn>() + " to same faction " + newFaction.ToStringSafe<Faction>());
      }
      else
      {
        Faction faction = this.Faction;
        this.guest?.SetGuestStatus((Faction) null);
        if (this.Spawned)
        {
          this.Map.mapPawns.DeRegisterPawn(this);
          this.Map.pawnDestinationReservationManager.ReleaseAllClaimedBy(this);
          this.Map.designationManager.RemoveAllDesignationsOn((Thing) this);
          this.Map.autoSlaughterManager.Notify_PawnChangedFaction();
        }
        if ((newFaction == Faction.OfPlayer || this.Faction == Faction.OfPlayer) && Current.ProgramState == ProgramState.Playing)
          Find.ColonistBar.MarkColonistsDirty();
        this.GetLord()?.Notify_PawnLost(this, PawnLostCondition.ChangedFaction);
        if (PawnUtility.IsFactionLeader(this))
        {
          Faction factionLeaderFaction = PawnUtility.GetFactionLeaderFaction(this);
          if (newFaction != factionLeaderFaction && !this.HasExtraHomeFaction(factionLeaderFaction) && !this.HasExtraMiniFaction(factionLeaderFaction))
            factionLeaderFaction.Notify_LeaderLost();
        }
        if (newFaction == Faction.OfPlayer && this.RaceProps.Humanlike && !this.IsQuestLodger())
          this.ChangeKind(newFaction.def.basicMemberKind);
        base.SetFaction(newFaction);
        PawnComponentsUtility.AddAndRemoveDynamicComponents(this);
        if (this.Faction != null && this.Faction.IsPlayer)
        {
          this.workSettings?.EnableAndInitialize();
          Find.StoryWatcher.watcherPopAdaptation.Notify_PawnEvent(this, PopAdaptationEvent.GainedColonist);
        }
        if (this.Drafted)
          this.drafter.Drafted = false;
        ReachabilityUtility.ClearCacheFor(this);
        this.health.surgeryBills.Clear();
        if (this.Spawned)
          this.Map.mapPawns.RegisterPawn(this);
        this.GenerateNecessaryName();
        this.playerSettings?.ResetMedicalCare();
        this.ClearMind(true);
        if (!this.Dead && this.needs.mood != null)
          this.needs.mood.thoughts.situational.Notify_SituationalThoughtsDirty();
        if (this.Spawned)
          this.Map.attackTargetsCache.UpdateTarget((IAttackTarget) this);
        Find.GameEnder.CheckOrUpdateGameOver();
        AddictionUtility.CheckDrugAddictionTeachOpportunity(this);
        this.needs?.AddOrRemoveNeedsAsAppropriate();
        this.playerSettings?.Notify_FactionChanged();
        this.relations?.Notify_ChangedFaction();
        if (this.IsNonMutantAnimal && newFaction == Faction.OfPlayer)
        {
          this.training.SetWantedRecursive(TrainableDefOf.Tameness, true);
          this.training.Train(TrainableDefOf.Tameness, recruiter, true);
          if (this.RaceProps.Roamer && this.mindState != null)
            this.mindState.lastStartRoamCooldownTick = new int?(Find.TickManager.TicksGame);
        }
        if (faction == Faction.OfPlayer)
          BillUtility.Notify_ColonistUnavailable(this);
        if (newFaction == Faction.OfPlayer)
        {
          Find.StoryWatcher.statsRecord.UpdateGreatestPopulation();
          Find.World.StoryState.RecordPopulationIncrease();
        }
        newFaction?.Notify_PawnJoined(this);
        this.Ideo?.Notify_MemberChangedFaction(this, faction, newFaction);
        this.ageTracker?.ResetAgeReversalDemand(Pawn_AgeTracker.AgeReversalReason.Recruited);
        this.roping?.BreakAllRopes();
        if (ModsConfig.BiotechActive)
          this.mechanitor?.Notify_ChangedFaction();
        this.creepjoiner?.Notify_ChangedFaction();
        if (faction == null)
          return;
        Find.FactionManager.Notify_PawnLeftFaction(faction);
      }
    }

    public void VerifyReservations(Job prevJob = null)
    {
      if (this.jobs == null || this.CurJob != null || this.jobs.jobQueue.Count > 0 || this.jobs.startingNewJob)
        return;
      bool flag = false;
      List<Map> maps = Find.Maps;
      for (int index = 0; index < maps.Count; ++index)
      {
        LocalTargetInfo localTargetInfo1 = maps[index].reservationManager.FirstReservationFor(this);
        if (localTargetInfo1.IsValid)
        {
          Log.ErrorOnce(string.Format("Reservation manager failed to clean up properly; {0} still reserving {1}, prev job: {2}", (object) this.ToStringSafe<Pawn>(), (object) localTargetInfo1.ToStringSafe<LocalTargetInfo>(), (object) prevJob), 97771429 ^ this.thingIDNumber);
          flag = true;
        }
        LocalTargetInfo localTargetInfo2 = maps[index].physicalInteractionReservationManager.FirstReservationFor(this);
        if (localTargetInfo2.IsValid)
        {
          Log.ErrorOnce("Physical interaction reservation manager failed to clean up properly; " + this.ToStringSafe<Pawn>() + " still reserving " + localTargetInfo2.ToStringSafe<LocalTargetInfo>() + ", prev job: {prevJob}", 19586765 ^ this.thingIDNumber);
          flag = true;
        }
        IAttackTarget attackTarget = maps[index].attackTargetReservationManager.FirstReservationFor(this);
        if (attackTarget != null)
        {
          Log.ErrorOnce("Attack target reservation manager failed to clean up properly; " + this.ToStringSafe<Pawn>() + " still reserving " + attackTarget.ToStringSafe<IAttackTarget>() + ", prev job: {prevJob}", 100495878 ^ this.thingIDNumber);
          flag = true;
        }
        IntVec3 intVec3 = maps[index].pawnDestinationReservationManager.FirstObsoleteReservationFor(this);
        if (intVec3.IsValid)
        {
          Job job = maps[index].pawnDestinationReservationManager.FirstObsoleteReservationJobFor(this);
          Log.ErrorOnce("Pawn destination reservation manager failed to clean up properly; " + this.ToStringSafe<Pawn>() + "/" + job.ToStringSafe<Job>() + "/" + job.def.ToStringSafe<JobDef>() + " still reserving " + intVec3.ToStringSafe<IntVec3>() + ", prev job: {prevJob}", 1958674 ^ this.thingIDNumber);
          flag = true;
        }
      }
      if (!flag)
        return;
      this.ClearAllReservations();
    }

    public void DropAndForbidEverything(bool keepInventoryAndEquipmentIfInBed = false, bool rememberPrimary = false)
    {
      if (this.kindDef.destroyGearOnDrop)
      {
        this.equipment.DestroyAllEquipment();
        this.apparel.DestroyAll();
      }
      if (this.InContainerEnclosed)
      {
        if (this.carryTracker?.CarriedThing != null)
          this.carryTracker.innerContainer.TryTransferToContainer(this.carryTracker.CarriedThing, this.holdingOwner);
        if (this.equipment?.Primary != null)
          this.equipment.TryTransferEquipmentToContainer(this.equipment.Primary, this.holdingOwner);
        this.inventory?.innerContainer.TryTransferAllToContainer(this.holdingOwner);
      }
      else
      {
        if (!this.SpawnedOrAnyParentSpawned)
          return;
        if (this.carryTracker?.CarriedThing != null)
          this.carryTracker.TryDropCarriedThing(this.PositionHeld, ThingPlaceMode.Near, out Thing _);
        if (keepInventoryAndEquipmentIfInBed && this.InBed())
          return;
        this.equipment?.DropAllEquipment(this.PositionHeld, rememberPrimary: rememberPrimary);
        if (this.inventory == null || this.inventory.innerContainer.TotalStackCount <= 0)
          return;
        this.inventory.DropAllNearPawn(this.PositionHeld, true);
      }
    }

    public Verb TryGetAttackVerb(Thing target, bool allowManualCastWeapons = false, bool allowTurrets = false)
    {
      if (this.equipment?.Primary != null && this.equipment.PrimaryEq.PrimaryVerb.Available() && ((!this.equipment.PrimaryEq.PrimaryVerb.verbProps.onlyManualCast ? 1 : (this.CurJob == null ? 0 : (this.CurJob.def != JobDefOf.Wait_Combat ? 1 : 0))) | (allowManualCastWeapons ? 1 : 0)) != 0)
        return this.equipment.PrimaryEq.PrimaryVerb;
      if (allowManualCastWeapons && this.apparel != null)
      {
        Verb firstApparelVerb = this.apparel.FirstApparelVerb;
        if (firstApparelVerb != null && firstApparelVerb.Available())
          return firstApparelVerb;
      }
      if (allowTurrets)
      {
        List<ThingComp> allComps = this.AllComps;
        for (int index = 0; index < allComps.Count; ++index)
        {
          if (allComps[index] is CompTurretGun compTurretGun && !compTurretGun.TurretDestroyed && compTurretGun.GunCompEq.PrimaryVerb.Available())
            return compTurretGun.GunCompEq.PrimaryVerb;
        }
      }
      return this.kindDef.canMeleeAttack ? this.meleeVerbs.TryGetMeleeVerb(target) : (Verb) null;
    }

    public bool TryStartAttack(LocalTargetInfo targ)
    {
      if (this.stances.FullBodyBusy || this.WorkTagIsDisabled(WorkTags.Violent))
        return false;
      bool allowManualCastWeapons = !this.IsColonist;
      Verb attackVerb = this.TryGetAttackVerb(targ.Thing, allowManualCastWeapons);
      return attackVerb != null && attackVerb.TryStartCastOn(attackVerb.verbProps.ai_RangedAlawaysShootGroundBelowTarget ? (LocalTargetInfo) targ.Cell : targ);
    }

    public override IEnumerable<Thing> ButcherProducts(Pawn butcher, float efficiency)
    {
      Pawn pawn = this;
      if (pawn.RaceProps.meatDef != null)
      {
        int num = GenMath.RoundRandom(pawn.GetStatValue(StatDefOf.MeatAmount) * efficiency);
        if (num > 0)
        {
          Thing thing = ThingMaker.MakeThing(pawn.RaceProps.meatDef);
          thing.stackCount = num;
          yield return thing;
        }
      }
      // ISSUE: reference to a compiler-generated method
      foreach (Thing thing in pawn.\u003C\u003En__1(butcher, efficiency))
        yield return thing;
      if (pawn.RaceProps.leatherDef != null)
      {
        int num = GenMath.RoundRandom(pawn.GetStatValue(StatDefOf.LeatherAmount) * efficiency);
        if (num > 0)
        {
          Thing thing = ThingMaker.MakeThing(pawn.RaceProps.leatherDef);
          thing.stackCount = num;
          yield return thing;
        }
      }
      if (!pawn.RaceProps.Humanlike)
      {
        PawnKindLifeStage lifeStage = pawn.ageTracker.CurKindLifeStage;
        if (lifeStage.butcherBodyPart != null && (pawn.gender == Gender.None || pawn.gender == Gender.Male && lifeStage.butcherBodyPart.allowMale || pawn.gender == Gender.Female && lifeStage.butcherBodyPart.allowFemale))
        {
          while (true)
          {
            BodyPartRecord partRecord = pawn.health.hediffSet.GetNotMissingParts().FirstOrDefault<BodyPartRecord>((Func<BodyPartRecord, bool>) (x => x.IsInGroup(lifeStage.butcherBodyPart.bodyPartGroup)));
            if (partRecord != null)
            {
              pawn.health.AddHediff(HediffMaker.MakeHediff(HediffDefOf.MissingBodyPart, pawn, partRecord));
              yield return ThingMaker.MakeThing(lifeStage.butcherBodyPart.thing ?? partRecord.def.spawnThingOnRemoved);
            }
            else
              break;
          }
        }
      }
    }

    public void HearClamor(Thing source, ClamorDef type)
    {
      if (this.Dead || this.Downed || this.Deathresting || this.IsSelfShutdown())
        return;
      if (type == ClamorDefOf.Movement || type == ClamorDefOf.BabyCry)
      {
        if (source is Pawn source1)
          this.CheckForDisturbedSleep(source1);
        this.NotifyLordOfClamor(source, type);
      }
      else if (type == ClamorDefOf.Harm)
      {
        if (this.Faction == Faction.OfPlayer || this.Awake() || this.Faction != source.Faction || this.HostFaction != null)
          return;
        this.mindState.canSleepTick = Find.TickManager.TicksGame + 1000;
        if (this.CurJob != null)
          this.jobs.EndCurrentJob(JobCondition.InterruptForced);
        this.NotifyLordOfClamor(source, type);
      }
      else if (type == ClamorDefOf.Construction)
      {
        if (this.Faction == Faction.OfPlayer || this.Awake() || this.Faction == source.Faction || this.HostFaction != null)
          return;
        this.mindState.canSleepTick = Find.TickManager.TicksGame + 1000;
        if (this.CurJob != null)
          this.jobs.EndCurrentJob(JobCondition.InterruptForced);
        this.NotifyLordOfClamor(source, type);
      }
      else if (type == ClamorDefOf.Ability)
      {
        if (this.Faction == Faction.OfPlayer || this.Faction == source.Faction || this.HostFaction != null)
          return;
        if (!this.Awake())
        {
          this.mindState.canSleepTick = Find.TickManager.TicksGame + 1000;
          if (this.CurJob != null)
            this.jobs.EndCurrentJob(JobCondition.InterruptForced);
        }
        this.NotifyLordOfClamor(source, type);
      }
      else
      {
        if (type != ClamorDefOf.Impact)
          return;
        this.mindState.Notify_ClamorImpact(source);
        if (this.CurJob != null && !this.Awake())
          this.jobs.EndCurrentJob(JobCondition.InterruptForced);
        this.NotifyLordOfClamor(source, type);
      }
    }

      {

        {
          allComps[index].Notify_Arrested(true);
          if (ModsConfig.AnomalyActive)
            this.creepjoiner?.Notify_Arrested(true);

    private void UpdatePyroVerbThought(Verb verb)
    {
      if (this.story == null || !this.story.traits.HasTrait(TraitDefOf.Pyromaniac) || !verb.IsIncendiary_Melee() && !verb.IsIncendiary_Ranged())
        return;
      LocalTargetInfo currentTarget = verb.CurrentTarget;
      if (currentTarget.Pawn != null)
      {
        currentTarget = verb.CurrentTarget;
        if (currentTarget.Pawn.Spawned)
        {
          currentTarget = verb.CurrentTarget;
          if (this.IsValidPyroThoughtTarget((Thing) currentTarget.Pawn))
          {
            this.needs?.mood?.thoughts?.memories?.TryGainMemory(ThoughtDefOf.PyroUsed);
            return;
          }
        }
      }
      currentTarget = verb.CurrentTarget;
      foreach (IntVec3 c in GenRadial.RadialCellsAround(currentTarget.Cell, verb.EffectiveRange, true))
      {
        if (c.InBounds(this.MapHeld))
        {
          foreach (Thing thing in c.GetThingList(this.MapHeld))
          {
            if (this.IsValidPyroThoughtTarget(thing))
            {
              Pawn_NeedsTracker needs = this.needs;
              if (needs != null)
              {
                Need_Mood mood = needs.mood;
                if (mood != null)
                {
                  ThoughtHandler thoughts = mood.thoughts;
                  if (thoughts != null)
                  {
                    MemoryThoughtHandler memories = thoughts.memories;
                    if (memories != null)
                    {
                      memories.TryGainMemory(ThoughtDefOf.PyroUsed);
                      break;
                    }
                    break;
                  }
                  break;
                }
                break;
              }
              break;
            }
          }
        }
      }
    }

    private bool IsValidPyroThoughtTarget(Thing thing)
    {
      return thing is Pawn pawn && !pawn.Downed && !pawn.IsPsychologicallyInvisible() && !pawn.Fogged() && pawn.HostileTo(Faction.OfPlayer);
    }

    public List<WorkTypeDef> GetDisabledWorkTypes(bool permanentOnly = false)
    {
      if (Scribe.mode != LoadSaveMode.Inactive)
      {
        this.cachedDisabledWorkTypesPermanent = (List<WorkTypeDef>) null;
        this.cachedDisabledWorkTypes = (List<WorkTypeDef>) null;
      }
      if (permanentOnly)
      {
        if (this.cachedDisabledWorkTypesPermanent == null)
        {
          this.cachedDisabledWorkTypesPermanent = new List<WorkTypeDef>();
          FillList(this.cachedDisabledWorkTypesPermanent);
        }
        return this.cachedDisabledWorkTypesPermanent;
      }
      if (this.cachedDisabledWorkTypes == null)
      {
        this.cachedDisabledWorkTypes = new List<WorkTypeDef>();
        FillList(this.cachedDisabledWorkTypes);
      }
      return this.cachedDisabledWorkTypes;

      void FillList(List<WorkTypeDef> list)
      {
        if (ModsConfig.AnomalyActive && this.IsMutant && this.mutant.HasTurned)
        {
          List<WorkTypeDef> defsListForReading = DefDatabase<WorkTypeDef>.AllDefsListForReading;
          for (int index = 0; index < defsListForReading.Count; ++index)
          {
            foreach (WorkTypeDef disabledWorkType in this.mutant.Def.DisabledWorkTypes)
            {
              if (!list.Contains(disabledWorkType))
                list.Add(disabledWorkType);
            }
          }
        }
        else
        {
          if (this.story != null && !this.IsSlave)
          {
            foreach (BackstoryDef allBackstory in this.story.AllBackstories)
            {
              foreach (WorkTypeDef disabledWorkType in allBackstory.DisabledWorkTypes)
              {
                if (!list.Contains(disabledWorkType))
                  list.Add(disabledWorkType);
              }
            }
            for (int index = 0; index < this.story.traits.allTraits.Count; ++index)
            {
              if (!this.story.traits.allTraits[index].Suppressed)
              {
                foreach (WorkTypeDef disabledWorkType in this.story.traits.allTraits[index].GetDisabledWorkTypes())
                {
                  if (!list.Contains(disabledWorkType))
                    list.Add(disabledWorkType);
                }
              }
            }
          }
          if (ModsConfig.BiotechActive && this.IsColonyMech)
          {
            List<WorkTypeDef> defsListForReading = DefDatabase<WorkTypeDef>.AllDefsListForReading;
            for (int index = 0; index < defsListForReading.Count; ++index)
            {
              if (!this.RaceProps.mechEnabledWorkTypes.Contains(defsListForReading[index]) && !list.Contains(defsListForReading[index]))
                list.Add(defsListForReading[index]);
            }
          }
          if (permanentOnly)
            return;
          if (this.health != null)
          {
            foreach (WorkTypeDef disabledWorkType in this.health.DisabledWorkTypes)
            {
              if (!list.Contains(disabledWorkType))
                list.Add(disabledWorkType);
            }
          }
          if (this.royalty != null && !this.IsSlave)
          {
            foreach (RoyalTitle royalTitle in this.royalty.AllTitlesForReading)
            {
              if (royalTitle.conceited)
              {
                foreach (WorkTypeDef disabledWorkType in royalTitle.def.DisabledWorkTypes)
                {
                  if (!list.Contains(disabledWorkType))
                    list.Add(disabledWorkType);
                }
              }
            }
          }
          if (ModsConfig.IdeologyActive && this.Ideo != null)
          {
            Precept_Role role = this.Ideo.GetRole(this);
            if (role != null)
            {
              foreach (WorkTypeDef disabledWorkType in role.DisabledWorkTypes)
              {
                if (!list.Contains(disabledWorkType))
                  list.Add(disabledWorkType);
              }
            }
          }
          if (ModsConfig.BiotechActive && this.genes != null)
          {
            foreach (Gene gene in this.genes.GenesListForReading)
            {
              foreach (WorkTypeDef disabledWorkType in gene.DisabledWorkTypes)
              {
                if (!list.Contains(disabledWorkType))
                  list.Add(disabledWorkType);
              }
            }
          }
          foreach (QuestPart_WorkDisabled partWorkDisabled in QuestUtility.GetWorkDisabledQuestPart(this))
          {
            foreach (WorkTypeDef disabledWorkType in partWorkDisabled.DisabledWorkTypes)
            {
              if (!list.Contains(disabledWorkType))
                list.Add(disabledWorkType);
            }
          }
          if (this.guest != null)
          {
            foreach (WorkTypeDef disabledWorkType in this.guest.GetDisabledWorkTypes())
            {
              if (!list.Contains(disabledWorkType))
                list.Add(disabledWorkType);
            }
          }
          for (int index = 0; index < this.RaceProps.lifeStageWorkSettings.Count; ++index)
          {
            LifeStageWorkSettings stageWorkSetting = this.RaceProps.lifeStageWorkSettings[index];
            if (stageWorkSetting.IsDisabled(this) && !list.Contains(stageWorkSetting.workType))
              list.Add(stageWorkSetting.workType);
          }
        }
      }
    }

        if (this.health?.hediffSet != null)
        {
          foreach (Hediff hediff in this.health.hediffSet.hediffs)
          {
            HediffStage curStage = hediff.CurStage;
            if (curStage != null)
              disabledWorkTags |= curStage.disabledWorkTags;
          }
        }
        foreach (QuestPart_WorkDisabled partWorkDisabled in QuestUtility.GetWorkDisabledQuestPart(this))
          disabledWorkTags |= partWorkDisabled.disabledWorkTags;
        if (this.IsMutant)
        {
          disabledWorkTags |= this.mutant.Def.workDisables;
          if (!this.mutant.IsPassive)
            disabledWorkTags &= ~WorkTags.Violent;
        }
        return disabledWorkTags;
      }
    }

    public bool WorkTagIsDisabled(WorkTags w) => (this.CombinedDisabledWorkTags & w) != 0;

    public override bool PreventPlayerSellingThingsNearby(out string reason)
    {
      if (this.Faction.HostileTo(Faction.OfPlayer) && this.HostFaction == null && !this.Downed && !this.InMentalState)
      {
        reason = (string) "Enemies".Translate();
        return true;
      }
      reason = (string) null;
      return false;
    }

    public void ChangeKind(PawnKindDef newKindDef)
    {
      if (this.kindDef == newKindDef)
        return;
      this.kindDef = newKindDef;
      if (!this.IsWildMan())
        return;
      this.mindState.WildManEverReachedOutside = false;
      ReachabilityUtility.ClearCacheFor(this);
    }
